# libadrenotools and its hooks (third_party/libadrenotools, BSD-2-Clause), which a game's host needs beside it to load a GPU driver, built as nx2android's
# dm/android/build.py builds them. The app puts them in the host's folder when a game is set to a driver of this app's (Store.driverRow)
set(AT third_party/libadrenotools)
set(LN ${AT}/lib/linkernsbypass)
function(adrenotools_lib name)
    add_library(${name} SHARED ${ARGN})
    target_include_directories(${name} PRIVATE ${AT}/include ${AT} ${LN})
    target_compile_options(${name} PRIVATE -O2 -fPIC -w)
    target_link_options(${name} PRIVATE -Wl,-z,max-page-size=16384 -static-libstdc++)
    target_link_libraries(${name} android log dl)
endfunction()
adrenotools_lib(adrenotools ${AT}/src/driver.cpp ${AT}/src/bcenabler.cpp ${LN}/android_linker_ns.cpp ${LN}/elf_soname_patcher.cpp)
adrenotools_lib(hook_impl ${AT}/src/hook/hook_impl.cpp ${LN}/android_linker_ns.cpp ${LN}/elf_soname_patcher.cpp)
target_compile_options(hook_impl PRIVATE -fvisibility=hidden)
foreach(hook main_hook file_redirect_hook gsl_alloc_hook)
    adrenotools_lib(${hook} ${AT}/src/hook/${hook}.c)
    target_link_options(${hook} PRIVATE -Wl,-z,global)
    target_link_libraries(${hook} hook_impl)
endforeach()
